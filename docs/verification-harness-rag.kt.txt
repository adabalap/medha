import java.nio.ByteBuffer
import java.nio.ByteOrder
import kotlin.math.sqrt

// ---- verbatim copies of the new pure logic ----
fun normalize(v: FloatArray): FloatArray {
    var s = 0.0; for (x in v) s += x.toDouble()*x
    val n = sqrt(s); if (n <= 1e-12) return v
    for (i in v.indices) v[i] = (v[i]/n).toFloat(); return v
}
fun dot(a: FloatArray, b: FloatArray): Double {
    if (a.size != b.size) return 0.0
    var s = 0.0; for (i in a.indices) s += a[i].toDouble()*b[i]; return s
}
fun encode(v: FloatArray): ByteArray {
    val bb = ByteBuffer.allocate(v.size*4).order(ByteOrder.LITTLE_ENDIAN)
    for (x in v) bb.putFloat(x); return bb.array()
}
fun decode(b: ByteArray): FloatArray {
    val bb = ByteBuffer.wrap(b).order(ByteOrder.LITTLE_ENDIAN)
    val out = FloatArray(b.size/4); for (i in out.indices) out[i] = bb.float; return out
}
const val QUERY_PREFIX = "task: search result | query: "
fun documentPrefix(title: String?) = "title: ${title?.takeIf{it.isNotBlank()} ?: "none"} | text: "

data class Hit(val text: String, val score: Double, val mode: String, val chunkId: Long = 0)
const val RRF_K = 60.0
fun fuse(dense: List<Hit>, lexical: List<Hit>, topK: Int): List<Hit> {
    val scores = HashMap<String, Double>(); val best = HashMap<String, Hit>()
    fun add(list: List<Hit>) = list.forEachIndexed { i, h ->
        scores[h.text] = (scores[h.text] ?: 0.0) + 1.0/(RRF_K+i+1); best.putIfAbsent(h.text, h) }
    add(dense); add(lexical)
    return scores.entries.sortedByDescending{it.value}.take(topK)
        .mapNotNull { e -> best[e.key]?.let { Hit(it.text, e.value, "hybrid", it.chunkId) } }
}

// simulated store row
data class Row(val id: Long, val text: String, val blob: ByteArray?, val model: String?, val dim: Int)
fun vectorCandidates(rows: List<Row>, model: String) =
    rows.filter { it.model == model && it.blob != null }

var pass=0; var fail=0
fun check(n: String, c: Boolean, d: String="") { if(c){pass++;println("  PASS $n")} else {fail++;println("  FAIL $n $d")} }

fun main() {
    println("== prefixes ==")
    check("query prefix exact", QUERY_PREFIX == "task: search result | query: ")
    check("doc prefix default title", documentPrefix(null) == "title: none | text: ")
    check("doc prefix with title", documentPrefix("Invoices") == "title: Invoices | text: ")
    check("blank title falls back to none", documentPrefix("  ") == "title: none | text: ")
    check("trailing space preserved", QUERY_PREFIX.endsWith(" ") && documentPrefix(null).endsWith(" "))

    println("\n== normalisation ==")
    val v = normalize(floatArrayOf(3f,4f))
    check("unit length", Math.abs(sqrt((v[0]*v[0]+v[1]*v[1]).toDouble())-1.0) < 1e-6)
    val z = normalize(floatArrayOf(0f,0f,0f))
    check("zero vector survives without NaN", z.none { it.isNaN() })
    val a = normalize(floatArrayOf(1f,2f,3f)); val b = normalize(floatArrayOf(1f,2f,3f))
    check("identical vectors cosine ~1", Math.abs(dot(a,b)-1.0) < 1e-5, dot(a,b).toString())
    val o1 = normalize(floatArrayOf(1f,0f)); val o2 = normalize(floatArrayOf(0f,1f))
    check("orthogonal cosine ~0", Math.abs(dot(o1,o2)) < 1e-6)
    check("dim mismatch returns 0 not crash", dot(floatArrayOf(1f), floatArrayOf(1f,2f)) == 0.0)

    println("\n== float32 LE blob codec ==")
    val orig = FloatArray(768) { (it % 17) * 0.13f - 1f }
    val round = decode(encode(orig))
    check("768-dim round trip exact", orig.contentEquals(round))
    check("blob is 3072 bytes", encode(orig).size == 3072, encode(orig).size.toString())
    check("little-endian byte order", encode(floatArrayOf(1.0f)).toList() == listOf<Byte>(0,0,-128,63),
        encode(floatArrayOf(1.0f)).toList().toString())
    check("blob smaller than CSV text", encode(orig).size < orig.joinToString(",").length,
        "${encode(orig).size} vs ${orig.joinToString(",").length}")

    println("\n== model-identity guard ==")
    val rows = listOf(
        Row(1,"current", encode(normalize(floatArrayOf(1f,0f))), "gemma@768", 768),
        Row(2,"stale-model", encode(normalize(floatArrayOf(1f,0f))), "gecko@768", 768),
        Row(3,"no-vector", null, null, 0))
    val cand = vectorCandidates(rows, "gemma@768")
    check("only same-space vectors returned", cand.map{it.id} == listOf(1L))
    check("stale model excluded", cand.none { it.model == "gecko@768" })
    check("unembedded excluded", cand.none { it.blob == null })

    println("\n== RRF fusion ==")
    val dense = listOf(Hit("paraphrase-match",0.91,"vector",1), Hit("both",0.80,"vector",2), Hit("d3",0.7,"vector",3))
    val lex   = listOf(Hit("exact-id-match",5.0,"lexical",4), Hit("both",4.0,"lexical",2), Hit("l3",3.0,"lexical",5))
    val f = fuse(dense, lex, 4)
    check("item in BOTH lists ranks first", f.first().text == "both", f.first().text)
    check("mode labelled hybrid", f.all { it.mode == "hybrid" })
    check("top-of-dense survives fusion", f.any { it.text == "paraphrase-match" })
    check("top-of-lexical survives fusion", f.any { it.text == "exact-id-match" })
    check("respects topK", f.size == 4)
    check("no duplicates", f.map{it.text}.toSet().size == f.size)
    // the SMS-organizer case: exact identifier only findable lexically
    val d2 = listOf(Hit("your order is on the way",0.88,"vector",1))
    val l2 = listOf(Hit("order AX-99213 dispatched",6.0,"lexical",2))
    val f2 = fuse(d2,l2,2)
    check("exact-identifier chunk retained in hybrid", f2.any{it.text.contains("AX-99213")})
    check("empty lists fuse to empty", fuse(emptyList(), emptyList(), 5).isEmpty())
    check("single-sided fusion works", fuse(dense, emptyList(), 2).size == 2)

    println("\n----------------------------------------")
    println("passed=$pass failed=$fail")
    if (fail>0) throw AssertionError("$fail failed")
}
